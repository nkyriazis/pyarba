# What the library needs to run. Deliberately unpinned: nothing here is
# version-sensitive except pybind11, and the constraint there is a floor,
# not a pin -- pybind11 < 2.12 segfaults converting a NumPy >= 2 array,
# which is the first thing every pyARBA entry point does.
#
# pybind11 is a build dependency; numpy, scipy and tqdm are the whole of
# the runtime. Importing any module under arba/ needs nothing else, and CI
# proves that on every push.
pybind11>=2.12
numpy
scipy
tqdm
